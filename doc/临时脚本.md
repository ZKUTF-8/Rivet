$exe = "C:\Program Files\OpenAI Codex\app\ChatGPT.exe"
$start = Get-Date

Get-Item $exe | Select-Object FullName,Length,LastWriteTime

$p = Start-Process $exe `
  -ArgumentList "--enable-logging=stderr","--v=1" `
  -PassThru

Start-Sleep 15
$p.Refresh()

[pscustomobject]@{
  ProcessId = $p.Id
  HasExited = $p.HasExited
  ExitCode  = if ($p.HasExited) { $p.ExitCode } else { $null }
}

Get-Process ChatGPT -ErrorAction SilentlyContinue |
  Select-Object Id,Path,StartTime

Get-WinEvent -FilterHashtable @{
  LogName   = "Application"
  StartTime = $start
} -ErrorAction SilentlyContinue |
  Where-Object {
    $_.ProviderName -match "Application Error|Windows Error Reporting" -or
    $_.Message -match "ChatGPT|OpenAI Codex"
  } |
  Select-Object TimeCreated,ProviderName,Id,LevelDisplayName,Message |
  Format-List

Get-MpThreatDetection -ErrorAction SilentlyContinue |
  Where-Object { $_.InitialDetectionTime -ge $start.AddMinutes(-10) } |
  Select-Object InitialDetectionTime,ThreatName,Resources,ActionSuccess |
  Format-List


Get-ChildItem "$env:LOCALAPPDATA\Codex\Logs" -File `
  -ErrorAction SilentlyContinue |
  Sort-Object LastWriteTime -Descending |
  Select-Object -First 10 FullName,Length,LastWriteTime
